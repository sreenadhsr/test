sequenceDiagram
    participant Client
    participant APIGEE
    participant AVS_API as AVS API Service
    participant Entitlement as Entitlement Service
    participant GUS as GUS Service API
    participant Fraud as Fraud Services API
    participant Audit as Audit Logging Service

    Client->>APIGEE: Send JSON Request (Company ID, User ID, Payload)
    APIGEE->>AVS_API: Forward Request with Headers
    AVS_API->>Entitlement: Validate Company & User Entitlements
    Entitlement-->>AVS_API: Entitlement Response
    alt Entitlement Failure
        AVS_API-->>APIGEE: Error Response (Trace ID)
        APIGEE-->>Client: Error Response (Trace ID)
    else Entitlement Success
        AVS_API->>AVS_API: Validate Request Payload
        alt Payload Invalid
            AVS_API-->>APIGEE: Error Response (Trace ID)
            APIGEE-->>Client: Error Response (Trace ID)
        else Payload Valid
            AVS_API->>GUS: Validate Routing Number
            GUS-->>AVS_API: Validation Result
            alt Routing Number Invalid
                AVS_API-->>APIGEE: Error Response (Trace ID)
                APIGEE-->>Client: Error Response (Trace ID)
            else Routing Number Valid
                AVS_API->>Fraud: Perform Fraud Check
                Fraud-->>AVS_API: Fraud Validation Result
                AVS_API->>Audit: Log Request/Response Details
                AVS_API-->>APIGEE: Success Response
                APIGEE-->>Client: Success Response
            end
        end
    end
